        j = 0;
        WHILE j < 3 DO
            j = j + 1;
        END

FUNCTION WhileIf()
    INT Counter;
    WHILE Counter < 10 DO
        IF Counter = 5 THEN
            RETURN Counter;
        END
        Counter = Counter + 1;
    END
